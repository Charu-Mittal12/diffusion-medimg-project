# torch / torchvision intentionally NOT pinned here — on Kaggle/Colab these
# come pre-installed and matched to the notebook's CUDA drivers. Reinstalling
# a specific pinned version on top can break GPU detection. If you need an
# exact version for strict reproducibility outside Kaggle/Colab, pin it
# separately in requirements.lock.txt instead, checked against
# `!python -c "import torch; print(torch.__version__)"` run FIRST.

# Core ML libraries. transformers is a RANGE, not an exact pin — 4.40.0
# depends on tokenizers 0.19.x, which has no prebuilt wheel for newer
# Python (e.g. 3.13) and is written in Rust, so it fails building from
# source without a Rust toolchain. A range lets pip pick a newer
# transformers whose tokenizers dependency already has a matching wheel.
diffusers==0.27.2
transformers>=4.40,<5
accelerate==0.29.3
peft==0.10.0
medmnist==3.0.1
wandb>=0.17.1,<1   # 0.17.0 exactly references np.float_, removed in NumPy 2.0 — needs a newer wandb release
protobuf>=5   # pinned loosely alongside wandb — an old protobuf can mismatch wandb's generated code (gencode/runtime version error)
lpips==0.1.4
datasets>=2.14,<3   # newer major versions dropped legacy script-based dataset loading, which the NIH-14 HF mirror still uses

# without system build tools.
